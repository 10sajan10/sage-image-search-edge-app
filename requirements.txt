# pywaggle WITHOUT the [vision] extra -- deliberately.
# The extra pulls opencv-python, the GUI build, which needs X11 libs (it fails
# with "ImportError: libxcb.so.1: cannot open shared object file"). We install
# the headless build below instead, which is the same cv2 module minus the GUI
# functions we never call. This mirrors what waggle/sage-thor-base does, but
# declaratively rather than by uninstalling and reinstalling opencv.
pywaggle==0.56.0

# What the [vision] extra would have provided, minus the GUI dependency.
opencv-python-headless>=4.10
ffmpeg-python>=0.2.0

# Image decoding/resizing for captioning + embedding (the populate path uses
# only Pillow; it never imports cv2).
pillow

# search.py only: the HTTP search API. main.py does not import these.
fastapi
uvicorn[standard]
pydantic

# jina-clip-v2 loads via trust_remote_code, and that remote code was written for
# transformers 4.x: it does `hasattr(torch, torch_dtype)` assuming a string,
# while transformers 5 passes a real torch.dtype and the config constructor dies
# with "TypeError: attribute name must be string, not 'torch.dtype'".
# This pin is load-bearing -- do not relax it without re-testing the loader.
transformers==4.53.3

# Required by jina-clip-v2's towers: timm for the EVA02 vision side, peft for
# the jina-embeddings-v3 text side (it declares LoRA adaptations).
timm
peft

# DO NOT add torch or numpy here.
#
# torch: the base image ships NVIDIA's Jetson build (2.13.0a0+...nv26.06),
# compiled with sm_110 kernels -- verified: arch_list includes sm_110, which is
# Thor's compute capability. PyPI's aarch64 wheels are generic/sbsa builds
# (Grace/GH200, sm_90) with no sm_110 kernels, and installing one would REPLACE
# the working build. It would fail silently: the image builds, then
# torch.cuda.is_available() is False and everything falls back to CPU.
#
# numpy: the base ships 2.1.0, which its torch works with (verified
# tensor.numpy() OK). Pinning is only needed on waggle/sage-thor-base:0.1.0,
# whose torch 2.8 rejects every numpy 2.x -- not an issue on this base.
#
# Qdrant and Ollama are reached over plain HTTP with urllib (stdlib), so no
# qdrant-client / SDK dependency is needed.
